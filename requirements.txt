# Use official lightweight Python image

# Prevent Python from writing .pyc files and enable unbuffered logs
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PORT=8001

WORKDIR /app

# Install dependencies first for better Docker layer caching
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy Tai blockchain core and node files
COPY tai.py node.py ./

# Create directory for persistent ledger/chain data
RUN mkdir -p /data

# Expose the default Tai node RPC port
EXPOSE 8001

# Start the node (ensure node.py binds to 0.0.0.0 and reads os.environ.get("PORT", 8001))
CMD ["python", "node.py"]
